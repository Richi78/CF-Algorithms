\subsubsection{Knapsack unbounded}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N * W)
// usa: como 0/1 pero cada item se puede tomar multiples veces
#include <bits/stdc++.h>
using namespace std;

int knapsackUnbounded(const vector<int>& w, const vector<int>& v, int W){
	int n = w.size();
	vector<int> dp(W + 1, 0);
	for(int i=0;i<n;i++){
		for(int j=w[i];j<=W;j++){
			dp[j] = max(dp[j], dp[j - w[i]] + v[i]);
		}
	}
	return dp[W];
}

int main(){
	vector<int> w = {2,3}, v = {3,4};
	cout << knapsackUnbounded(w, v, 7) << "\n";  // 11 (3+4+4)
	return 0;
}
\end{lstlisting}
