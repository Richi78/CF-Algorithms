\subsubsection{Tree DP (Reroot aditivo)}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N)
// usa: para cada nodo, calcular alguna propiedad asumiendo que ese nodo es la raiz
#include <bits/stdc++.h>
using namespace std;

struct TreeDP{
	int n;
	vector<vector<int>> adj;
	vector<long long> dp, up, ans;
	vector<int> sz;
	TreeDP(int n): n(n){
		adj.assign(n, {});
		dp.assign(n, 0);
		up.assign(n, 0);
		ans.assign(n, 0);
		sz.assign(n, 0);
	}
	void addEdge(int u, int v){
		adj[u].push_back(v);
		adj[v].push_back(u);
	}
	// 1st pass: compute dp[v] as the answer for subtree of v (rooted at 0)
	void dfs1(int v, int p){
		sz[v] = 1;
		for(int u : adj[v]){
			if(u == p) continue;
			dfs1(u, v);
			sz[v] += sz[u];
			dp[v] += dp[u] + sz[u]; // example - sum of distances from u
		}
	}
	// 2nd pass: reroot - propagate contributions
	void dfs2(int v, int p){
		ans[v] = dp[v] + up[v];
		for(int u : adj[v]){
			if(u == p) continue;
			// when rerooting at u, u's contribution comes out, v's grows
			up[u] = up[v] + dp[v] - (dp[u] + sz[u]) + (n - sz[u]);
			dfs2(u, v);
		}
	}
	vector<long long> solve(int root = 0){
		dfs1(root, -1);
		dfs2(root, -1);
		return ans;
	}
};
\end{lstlisting}

\subsubsection{Tree DP (Reroot prefijos y sufijos)}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N)
// Usa pref/suf cuando la operacion NO es invertible
// (ej. mod compuesto, division por 0, max, gcd)
const int MOD = 1e9 + 7;
vector<vector<int>> adj;
vector<long long> dp, up, ans;

void dfs(int node, int par){
  dp[node] = 1;
  for(auto adjN : adj[node]) if(adjN != par){
    dfs(adjN, node);
    dp[node] = dp[node] * ((dp[adjN] + 1) % MOD) % MOD;
  }
}

void dfs2(int node, int par){
  vector<int> children;
  for(auto adjN : adj[node]) if(adjN != par){
    children.push_back(adjN);
  }
  int sz = children.size();
  vector<long long> pref(sz + 1, 1), suf(sz + 1, 1);
  for(int i = 0; i < sz; i++){
    pref[i + 1] = pref[i] * ((dp[children[i]] + 1) % MOD) % MOD;
  }
  for(int i = sz - 1; i >= 0; i--){
    suf[i] = suf[i + 1] * ((dp[children[i]] + 1) % MOD) % MOD;
  }
  for(int i = 0; i < sz; i++){
    int child = children[i];
    long long sib = pref[i] * suf[i + 1] % MOD;
    up[child] = sib * ((up[node] + 1) % MOD) % MOD;
    ans[child] = dp[child] * ((up[child] + 1) % MOD) % MOD;
    dfs2(child, node);
  }
}

void solve(int root = 0){
  dfs(root, -1);
  up[root] = 0;
  ans[root] = dp[root];
  dfs2(root, -1);
}
\end{lstlisting}
