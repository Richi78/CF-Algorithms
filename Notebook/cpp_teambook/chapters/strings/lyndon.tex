\subsubsection{Lyndon factorization (Duval)}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N)
// usa: descomponer un string en factores de Lyndon
#include <bits/stdc++.h>
using namespace std;

vector<int> lyndonFactorization(const string& s){
	// returns ending positions of each Lyndon factor (0-indexed)
	int n = s.size();
	vector<int> cuts;
	int i = 0;
	while(i < n){
		int j = i, k = i + 1;
		while(k < n && s[j] <= s[k]){
			if(s[j] < s[k]) j = i;
			else j++;
			k++;
		}
		int len = k - j;
		while(i <= j){
			cuts.push_back(i + len - 1);
			i += len;
		}
	}
	return cuts;
}

int main(){
	string s = "ababa";
	auto cuts = lyndonFactorization(s);
	for(int c : cuts) cout << c << " ";  // 0 2 4
	cout << "\n";
	return 0;
}
\end{lstlisting}
