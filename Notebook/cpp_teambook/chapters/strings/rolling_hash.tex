\subsubsection{Rolling Hash (doble modulo)}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: precompute O(N), getHash O(1)  (N = |s|)
// usar: comparar substrings, palindromos, longest common substring,
//       contar substrings distintos, hashing de paths en trees, etc.
// cuidado: doble hash reduce colisiones a casi cero pero no las elimina
struct pair_hash {
	size_t operator()(const pair<int,int>& p) const {
		return ((uint64_t)p.first << 32) ^ (uint64_t)p.second;
	}
};
struct H{
	vector<pair<int,int>> h,p;
	const int MOD1=998244353;
	const int MOD2=1e9+7;
	const int A=31;
	string s;
	H(string &a){
		s=a;
		int n=s.size();
		h.resize(n+1); p.resize(n+1,{1,1});
		for(int i=1 ; i<=n ; i++){
			h[i].F=((h[i-1].F*A)%MOD1 + s[i-1])%MOD1;
			h[i].S=((h[i-1].S*A)%MOD2 + s[i-1])%MOD2;
			p[i].F=(p[i-1].F*A)%MOD1;
			p[i].S=(p[i-1].S*A)%MOD2;
		}
	}
	pair<int,int> getHash(int l, int r){
		pair<int,int> ans;
		ans.F=(h[r].F-(h[l-1].F*p[r-l+1].F)%MOD1)%MOD1;
		ans.S=(h[r].S-(h[l-1].S*p[r-l+1].S)%MOD2)%MOD2;
		ans.F=(ans.F+MOD1)%MOD1;
		ans.S=(ans.S+MOD2)%MOD2;
		return ans;
	}
};
// unordered_set<pair<int,int>, pair_hash> st;

int main(){
	string s = "banana";
	H h(s);
	// comparar substrings en O(1) por hash (1-indexed, inclusivo)
	auto a = h.getHash(1, 3); // "ban"
	auto b = h.getHash(4, 6); // "ana"
	cout << (a == b ? "iguales" : "distintos") << "\n";
	return 0;
}
\end{lstlisting}
