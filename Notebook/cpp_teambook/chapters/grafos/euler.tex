\subsubsection{Euler tour / Hierholzer (camino y circuito)}

\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N + M) - Algoritmo de Hierholzer
// usa: existencia y construccion de camino / circuito euleriano
//     (visita cada ARISTA exactamente una vez; admite multigrafos)
#include <bits/stdc++.h>
using namespace std;

// ---------- Condiciones necesarias y suficientes ----------
//   NO DIRIGIDO, conexo:
//     - circuito euleriano: todos los vertices tienen grado par
//     - camino  euleriano: exactamente 0 o 2 vertices con grado impar
//                            (si son 2, el camino va de uno al otro)
//   DIRIGIDO, debilmente conexo sobre el subgrafo con aristas:
//     - circuito euleriano: indeg(v) == outdeg(v) para todo v
//     - camino  euleriano: existe s con outdeg(s) -indeg(s) = 1,
//                          existe t con indeg(t) - outdeg(t) = 1,
//                          y el resto cumple indeg == outdeg

int main(){
	int n,m; cin >> n >> m;
	vector<pair<int,int>> adj[n]; // no dirigido necesitamos par para guardar id
	// vector<int> indeg(n) , outdeg(n);
	vector<int> deg(n);
	for(int i=0 ; i<m ; i++){
		int u,v; cin >> u >> v; u--; v--;
		// si es dirigido solo usar uno y ya no seria necesario guardar id
		adj[u].emplace_back(v,i);
		adj[v].emplace_back(u,i);
		deg[u]++; deg[v]++; // si es dirigido llenar indeg y outdeg
	}
	int start=0 , end=-1;
	int odd=0;
	for(int i=0 ; i<n ; i++){
		// contar odd, asignar start, end , revisar condiciones de existencia
		if(deg[i]&1){
			start=i; odd++; 
		}
	}
	if(odd!=0 && odd!=2){
		// no hay camino ni circuito
		return 0;
	}

	stack<int> st;
	// solo si es no dirigido
	vector<bool> vis_edges(m);
	vector<int> path; // o circuit
	st.push(start);
	while(!st.empty()){
		int node = st.top();
		// si es no dirigido lo necesitamos
		while(!adj[node].empty() && vis_edges[adj[node].back().second]){
			adj[node].pop_back();
		}
		if(!adj[node].empty()){
			auto [adjN,id] = adj[node].back();
			adj[node].pop_back();
			st.push(adjN);
			vis_edges[id]=true;
		}else{
			path.push_back(node);
			st.pop();
		}
	}
	reverse(path.begin() , path.end());

	// check si va por todos los edges
	if(path.size() != m+1){
		// no pasa por todo
	}
	
	for(auto &x : path) cout<< x <<" ";
	cout<<"\n";
}
\end{lstlisting}
