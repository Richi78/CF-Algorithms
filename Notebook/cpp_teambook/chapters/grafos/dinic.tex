\subsubsection{Dinic (max flow)}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N^2 * M) general, O(sqrt(N) * M) for bipartite
// usa: max flow eficiente, base para muchos problemas
#include <bits/stdc++.h>
using namespace std;

const long long INF = 1e18;

struct Dinic{
	struct Edge{ int adjN, rev; long long cap; };
	int n, s, t;
	vector<vector<Edge>> adj;
	vector<int> level, it;
	Dinic(int n, int s, int t): n(n), s(s), t(t){
		adj.assign(n, {});
		level.assign(n, -1);
		it.assign(n, 0);
	}
	void addEdge(int u, int v, long long cap){
		Edge a{v, (int)adj[v].size(), cap};
		Edge b{u, (int)adj[u].size(), 0};
		adj[u].push_back(a);
		adj[v].push_back(b);
	}
	bool bfs(){
		fill(level.begin(), level.end(), -1);
		queue<int> q;
		level[s] = 0; q.push(s);
		while(!q.empty()){
			int node = q.front(); q.pop();
			for(const auto& [adjN, rev, cap] : adj[node]){
				if(cap > 0 && level[adjN] == -1){
					level[adjN] = level[node] + 1;
					q.push(adjN);
				}
			}
		}
		return level[t] != -1;
	}
	long long dfs(int node, long long pushed){
		if(pushed == 0) return 0;
		if(node == t) return pushed;
		for(int& i = it[node]; i < (int)adj[node].size(); i++){
			auto& [adjN, rev, cap] = adj[node][i];
			if(cap > 0 && level[adjN] == level[node] + 1){
				long long tr = dfs(adjN, min(pushed, cap));
				if(tr == 0) continue;
				cap -= tr;
				adj[adjN][rev].cap += tr;
				return tr;
			}
		}
		return 0;
	}
	long long maxflow(){
		long long flow = 0;
		while(bfs()){
			fill(it.begin(), it.end(), 0);
			while(long long pushed = dfs(s, INF)){
				flow += pushed;
			}
		}
		return flow;
	}
};

int main(){
	Dinic dinic(4, 0, 3);
	dinic.addEdge(0,1,10);
	dinic.addEdge(0,2,5);
	dinic.addEdge(1,2,15);
	dinic.addEdge(1,3,10);
	dinic.addEdge(2,3,10);
	cout << "max flow = " << dinic.maxflow() << "\n";  // 15
	return 0;
}
\end{lstlisting}
