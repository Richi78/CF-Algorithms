\subsubsection{Tarjan SCC}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N + M)
// usa: encontrar componentes fuertemente conexos
#include <bits/stdc++.h>
using namespace std;

struct TarjanSCC{
	int n, timer, numComps;
	vector<int> comp, low, disc, st;
	vector<bool> inStack;
	const vector<vector<int>>& adj;
	TarjanSCC(int n, const vector<vector<int>>& adj): n(n), adj(adj){
		comp.assign(n, -1);
		low.assign(n, 0);
		disc.assign(n, 0);
		inStack.assign(n, false);
		timer = 0;
		numComps = 0;
		for(int i=0;i<n;i++) if(disc[i] == 0) dfs(i);
	}
	void dfs(int v){
		disc[v] = low[v] = ++timer;
		st.push_back(v);
		inStack[v] = true;
		for(int u : adj[v]){
			if(disc[u] == 0){
				dfs(u);
				low[v] = min(low[v], low[u]);
			}else if(inStack[u]){
				low[v] = min(low[v], disc[u]);
			}
		}
		if(low[v] == disc[v]){
			while(true){
				int u = st.back(); st.pop_back();
				inStack[u] = false;
				comp[u] = numComps;
				if(u == v) break;
			}
			numComps++;
		}
	}
};

int main(){
	vector<vector<int>> adj = {{1}, {2,3}, {0}, {4}, {}};
	TarjanSCC scc(5, adj);
	cout << "#SCCs = " << scc.numComps << "\n";  // 3
	return 0;
}
\end{lstlisting}
