\subsubsection{Tarjan SCC}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N + M)
// usa: encontrar componentes fuertemente conexos
#include <bits/stdc++.h>
using namespace std;

struct TarjanSCC{
	int n, timer, numComps;
	vector<int> comp, low, disc, st;
	vector<bool> inStack;
	const vector<vector<int>>& adj;
	TarjanSCC(int n, const vector<vector<int>>& adj): n(n), adj(adj){
		comp.assign(n, -1);
		low.assign(n, 0);
		disc.assign(n, 0);
		inStack.assign(n, false);
		timer = 0;
		numComps = 0;
		for(int i=0;i<n;i++) if(disc[i] == 0) dfs(i);
	}
	void dfs(int node){
		disc[node] = low[node] = ++timer;
		st.push_back(node);
		inStack[node] = true;
		for(auto adjN : adj[node]){
			if(disc[adjN] == 0){
				dfs(adjN);
				low[node] = min(low[node], low[adjN]);
			}else if(inStack[adjN]){
				low[node] = min(low[node], disc[adjN]);
			}
		}
		if(low[node] == disc[node]){
			while(true){
				int cur = st.back(); st.pop_back();
				inStack[cur] = false;
				comp[cur] = numComps;
				if(cur == node) break;
			}
			numComps++;
		}
	}
};

int main(){
	vector<vector<int>> adj = {{1}, {2,3}, {0}, {4}, {}};
	TarjanSCC scc(5, adj);
	cout << "#SCCs = " << scc.numComps << "\n";  // 3
	return 0;
}
\end{lstlisting}
