\subsubsection{Tarjan SCC}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N + M)
// usa: encuentra componentes fuertemente conexas (SCC) en una sola pasada de DFS
vector<vector<int>> adj;
vector<int> tin, low, scc;
vector<bool> inStack;
stack<int> st;
int t = 0, scc_number = 0;

void dfs(int node){
	tin[node] = low[node] = ++t;
	st.push(node);
	inStack[node] = true;

	for(auto adjN : adj[node]){
		if(!tin[adjN]){
			dfs(adjN);
			low[node] = min(low[node], low[adjN]);
		}else if(inStack[adjN]){
			low[node] = min(low[node], tin[adjN]);
		}
	}

	if(low[node] == tin[node]){
		scc_number++;
		while(true){
			int cur = st.top(); st.pop();
			inStack[cur] = false;
			scc[cur] = scc_number;
			if(cur == node) break;
		}
	}
}

void solve(){
	int n, m; cin >> n >> m;
	adj.assign(n + 1, {});
	tin.assign(n + 1, 0);
	low.assign(n + 1, 0);
	scc.assign(n + 1, 0);
	inStack.assign(n + 1, false);
	while(!st.empty()) st.pop();
	t = 0; scc_number = 0;

	for(int i = 0; i < m; i++){
		int u, v; cin >> u >> v;
		adj[u].push_back(v);
	}

	for(int i = 1; i <= n; i++){
		if(!tin[i]) dfs(i);
	}
}
\end{lstlisting}
