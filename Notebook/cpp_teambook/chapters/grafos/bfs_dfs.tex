\subsubsection{DFS / BFS / componentes}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N + M)
// usa: recorrido, conteo de componentes
#include <bits/stdc++.h>
using namespace std;

struct Graph{
	int n;
	vector<vector<int>> adj;
	Graph(int n): n(n), adj(n){}
	void addEdge(int u, int v, bool undirected = true){
		adj[u].push_back(v);
		if(undirected) adj[v].push_back(u);
	}
	vector<int> bfs(int start){
		vector<int> dist(n, -1);
		queue<int> q;
		dist[start] = 0; q.push(start);
		while(!q.empty()){
			int node = q.front(); q.pop();
			for(auto adjN : adj[node]) if(dist[adjN] == -1){
				dist[adjN] = dist[node] + 1;
				q.push(adjN);
			}
		}
		return dist;
	}
	void dfsRec(int node, vector<bool>& vis, vector<int>& order){
		vis[node] = true;
		order.push_back(node);
		for(auto adjN : adj[node]) if(!vis[adjN]) dfsRec(adjN, vis, order);
	}
	vector<int> dfs(int start){
		vector<int> order;
		vector<bool> vis(n, false);
		dfsRec(start, vis, order);
		return order;
	}
	int countComponents(){
		vector<bool> vis(n, false);
		vector<int> order;
		int cnt = 0;
		for(int i=0 ; i<n ; i++){
			if(!vis[i]){
				dfsRec(i, vis, order);
				cnt++;
			}
		}
		return cnt;
	}
};

int main(){
	Graph g(5);
	g.addEdge(0,1); g.addEdge(1,2); g.addEdge(3,4);  // 2 components
	cout << "components: " << g.countComponents() << "\n";  // 2
	auto dist = g.bfs(0);
	for(int d : dist) cout << d << " ";
	cout << "\n";
	return 0;
}
\end{lstlisting}
