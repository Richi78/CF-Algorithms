\subsubsection{Bellman-Ford}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N * M)
// usa: camino minimo desde s; permite pesos negativos; detecta ciclo negativo alcanzable desde s en la relajacion extra
vector<int> ford(int n, vector<array<int,3>> edges, int s){
	// u v w
	vector<int> dist(n,1e18);
	dist[s]=0;
	for(int i=0 ; i<n-1 ; i++){ // n-1 relaxations
		for(auto x : edges){
			int u=x[0] , v=x[1] , w=x[2];
			if(dist[u] != 1e18 && dist[u]+w < dist[v]){
				dist[v]=dist[u]+w;
			}
		}
	}
	// detect negative cycle one more relaxation
	for(auto x : edges){
		int u=x[0] , v=x[1] , w=x[2];
		if(dist[u] != 1e18 && dist[u]+w < dist[v]){
			// there is a negative cycle
			}
		}
	return dist;
}
\end{lstlisting}
