\subsubsection{2-SAT (implication graph + SCC)}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N + M)
// usa: formulas booleanas en CNF con clausulas de 2 literales
// Clausulas comunes (not x = x^1):
// - A or B   : adj[a^1].push_back(b),   adj[b^1].push_back(a)
// - A xor B  : (A v B) ^ (~A v ~B)  [solo 1 es true, A != B]
//              adj[a^1].push_back(b),   adj[b^1].push_back(a)
//              adj[a].push_back(b^1),   adj[b].push_back(a^1)
// - A <=> B  : (A => B) ^ (B => A)    [ambos iguales, A == B]
//              adj[a].push_back(b),     adj[b^1].push_back(a^1)
//              adj[b].push_back(a),     adj[a^1].push_back(b^1)
// - Forzar A : adj[a^1].push_back(a)  (~A => A obliga a que A sea true)
// - Forzar ~A: adj[a].push_back(a^1)  (A => ~A obliga a que A sea false)
#include <bits/stdc++.h>
using namespace std;

vector<bool> vis;
vector<int> scc;
stack<int> st;
int scc_number=-1;

int main(){
	int n,m; cin >> n >> m;
	vector<int> adj[2*m]; // doble de aristas, par true, impar false
	for(int i=0 ; i<m ; i++){
		char t1,t2; int v1,v2; cin >> t1 >> v1 >> t2 >> v2;
		v1--; v2--;
		int a=2*v1 + (t1=='-');
		int b=2*v2 + (t2=='-');
		adj[a^1].push_back(b);
		adj[b^1].push_back(a);
	}

	vis.resize(2*m);
	for(int i=0 ; i<2*m ; i++) if(!vis[i]) dfs1(i,adj);

	vector<int> adjT[2*m];
	for(int i=0 ; i<2*m ; i++){
		vis[i]=false;
		for(int adjN : adj[i]) adjT[adjN].push_back(i);
	}

	scc.resize(2*m);
	while(!st.empty()){
		int node = st.top();
		st.pop();
		if(!vis[node]){
			scc_number++;
			dfs2(node,adjT);
		}
	}

	vector<char> ans(m);
	for(int i=0 ; i<m ; i++){
		if(scc[2*i] == scc[2*i+1]){
			// contradiccion
			return 0;
		}
		if(scc[2*i] > scc[2*i+1]) ans[i]='+';
		else ans[i]='-';
	}
	
	for(auto &x : ans) cout<< x <<" ";
	cout<<"\n";

	return 0;
}
\end{lstlisting}
