\subsubsection{Kosaraju SCC}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N + M)
// usa: encuentra componentes fuertemente conexas (SCC) usando dos DFS: una por orden de salida y otra sobre el grafo transpuesto
vector<vector<pair<int,int>>> adj, adjT, adjCond;
vector<bool> vis;
vector<int> scc;
stack<int> st;
int scc_number = 0;

void dfs(int node){
	vis[node] = 1;
	for(auto [adjN, adjW] : adj[node]){
		if(!vis[adjN]) dfs(adjN);
	}
	st.push(node);
}

void dfs2(int node){
	vis[node] = 1;
	scc[node] = scc_number;
	for(auto [adjN, adjW] : adjT[node]){
		if(!vis[adjN]) dfs2(adjN);
	}
}

void solve(){
	int n, m; cin >> n >> m;
	adj.assign(n, {});
	adjT.assign(n, {});
	vis.assign(n, false);
	scc.assign(n, 0);
	scc_number = 0;

	for(int i = 0; i < m; i++){
		int u, v, w; cin >> u >> v >> w;
		u--; v--;
		adj[u].emplace_back(v, w);
		adjT[v].emplace_back(u, w);
	}

	for(int i = 0; i < n; i++){
		if(!vis[i]) dfs(i);
	}

	vis.assign(n, false);
	while(!st.empty()){
		int node = st.top(); st.pop();
		if(!vis[node]){
			scc_number++;
			dfs2(node);
		}
	}

	adjCond.assign(scc_number, {});
	for(int i = 0; i < n; i++){
		for(auto [adjN, adjW] : adj[i]){
			if(scc[i] != scc[adjN]){
				adjCond[scc[i] - 1].emplace_back(scc[adjN] - 1, adjW);
			}
		}
	}
}
\end{lstlisting}
