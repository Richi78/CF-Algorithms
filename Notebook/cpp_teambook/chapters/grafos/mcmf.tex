\subsubsection{MCMF con SPFA (Simple / Costos Negativos)}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(F * V * E) peor caso. En la practica suele ser rapido.
// CUANDO USAR:
// - Grafos pequenos o medianos (V <= 200, E <= 1000).
// - Grafos con costos iniciales negativos (sin ciclos neg).
// - Facil y rapido de tipear en contest (sin potenciales).
#include <bits/stdc++.h>
using namespace std;

const long long INF = 1e18;

struct MCMF_SPFA {
  struct Edge { int to, rev; long long cap, cost; };
  int n;
  vector<vector<Edge>> g;
  MCMF_SPFA(int n): n(n), g(n) {}

  void add_edge(int u, int v, long long cap, long long cost) {
    g[u].push_back({v, (int)g[v].size(), cap, cost});
    g[v].push_back({u, (int)g[u].size() - 1, 0, -cost});
  }

  // Retorna {max_flow, min_cost}
  pair<long long, long long> min_cost_flow(int s, int t) {
    long long flow = 0, cost = 0;
    vector<long long> dist(n);
    vector<int> pv(n), pe(n);
    vector<bool> inq(n);
    queue<int> q;

    while (true) {
      fill(dist.begin(), dist.end(), INF);
      fill(inq.begin(), inq.end(), false);
      dist[s] = 0;
      q.push(s);
      inq[s] = true;

      while (!q.empty()) {
        int u = q.front(); q.pop();
        inq[u] = false;
        for (int i = 0; i < (int)g[u].size(); i++) {
          auto &e = g[u][i];
          if (e.cap > 0 && dist[u] + e.cost < dist[e.to]) {
            dist[e.to] = dist[u] + e.cost;
            pv[e.to] = u;
            pe[e.to] = i;
            if (!inq[e.to]) {
              inq[e.to] = true;
              q.push(e.to);
            }
          }
        }
      }

      if (dist[t] == INF) break; // no hay mas caminos

      long long push = INF;
      for (int v = t; v != s; v = pv[v]) {
        push = min(push, g[pv[v]][pe[v]].cap);
      }
      for (int v = t; v != s; v = pv[v]) {
        auto &e = g[pv[v]][pe[v]];
        e.cap -= push;
        g[v][e.rev].cap += push;
        cost += push * e.cost;
      }
      flow += push;
    }
    return {flow, cost};
  }
};
\end{lstlisting}

\subsubsection{MCMF con Potenciales de Johnson (Dijkstra / Rapido)}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(F * (V + E) log V). Con SPFA inicial: +O(V * E).
// CUANDO USAR:
// - Grafos grandes (V hasta 2000-5000, E hasta 20000+).
// - Flujo grande F o limites de tiempo estrictos (evita TLE).
// - Garantia contra grafos "SPFA-killer".
// NOTA: Si hay costos iniciales < 0, se corre SPFA una sola
// vez al inicio para fijar los potenciales h[].
#include <bits/stdc++.h>
using namespace std;

const long long INF = 1e18;

struct MCMF_Pot {
  struct Edge { int to, rev; long long cap, cost; };
  int n;
  vector<vector<Edge>> g;
  MCMF_Pot(int n): n(n), g(n) {}

  void add_edge(int u, int v, long long cap, long long cost) {
    g[u].push_back({v, (int)g[v].size(), cap, cost});
    g[v].push_back({u, (int)g[u].size() - 1, 0, -cost});
  }

  // Opcional: solo si hay aristas con costo inicial < 0
  void init_potentials_spfa(int s, vector<long long> &h) {
    fill(h.begin(), h.end(), INF);
    vector<bool> inq(n, false);
    queue<int> q;
    h[s] = 0; q.push(s); inq[s] = true;
    while (!q.empty()) {
      int u = q.front(); q.pop();
      inq[u] = false;
      for (auto &e : g[u]) {
        if (e.cap > 0 && h[u] != INF &&
            h[u] + e.cost < h[e.to]) {
          h[e.to] = h[u] + e.cost;
          if (!inq[e.to]) {
            inq[e.to] = true;
            q.push(e.to);
          }
        }
      }
    }
    for (int i = 0; i < n; i++) if (h[i] == INF) h[i] = 0;
  }

  // Retorna {max_flow, min_cost}
  pair<long long, long long> min_cost_flow(int s, int t) {
    long long flow = 0, cost = 0;
    vector<long long> h(n, 0); // potenciales
    // Descomentar si hay costos iniciales negativos:
    // init_potentials_spfa(s, h);

    vector<long long> dist(n);
    vector<int> pv(n), pe(n);
    using P = pair<long long, int>;

    while (true) {
      priority_queue<P, vector<P>, greater<P>> pq;
      fill(dist.begin(), dist.end(), INF);
      dist[s] = 0;
      pq.push({0, s});

      while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (d > dist[u]) continue;
        for (int i = 0; i < (int)g[u].size(); i++) {
          auto &e = g[u][i];
          // Costo reducido no-negativo: cost + h[u] - h[e.to]
          long long red = e.cost + h[u] - h[e.to];
          if (e.cap > 0 && dist[u] + red < dist[e.to]) {
            dist[e.to] = dist[u] + red;
            pv[e.to] = u;
            pe[e.to] = i;
            pq.push({dist[e.to], e.to});
          }
        }
      }

      if (dist[t] == INF) break; // no hay mas caminos

      // Actualizar potenciales (clamping a dist[t])
      for (int i = 0; i < n; i++) {
        h[i] += min(dist[i], dist[t]);
      }

      long long push = INF;
      for (int v = t; v != s; v = pv[v]) {
        push = min(push, g[pv[v]][pe[v]].cap);
      }
      for (int v = t; v != s; v = pv[v]) {
        auto &e = g[pv[v]][pe[v]];
        e.cap -= push;
        g[v][e.rev].cap += push;
        cost += push * e.cost;
      }
      flow += push;
    }
    return {flow, cost};
  }
};
\end{lstlisting}
