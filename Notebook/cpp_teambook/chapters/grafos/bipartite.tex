\subsubsection{Bipartite check + 2-coloring}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N + M)
// usa: chequear si un grafo es bipartito y asignar colores
#include <bits/stdc++.h>
using namespace std;

pair<bool, vector<int>> isBipartite(int n, const vector<vector<int>>& adj){
	vector<int> color(n, -1);
	for(int i=0 ; i<n ; i++){
		if(color[i] != -1) continue;
		queue<int> q;
		color[i] = 0; q.push(i);
		while(!q.empty()){
			int node = q.front(); q.pop();
			for(auto adjN : adj[node]){
				if(color[adjN] == -1){ color[adjN] = color[node] ^ 1; q.push(adjN); }
				else if(color[adjN] == color[node]) return {false, {}};
			}
		}
	}
	return {true, color};
}

int main(){
	vector<vector<int>> adj = {{1}, {0,2}, {1,3}, {2}};  // path of 4: bipartite
	auto [ok, color] = isBipartite(4, adj);
	cout << (ok ? "bipartite" : "not bipartite") << "\n";
	return 0;
}
\end{lstlisting}
