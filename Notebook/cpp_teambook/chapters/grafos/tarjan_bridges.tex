\subsubsection{Tarjan (puentes)}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N + M)
// usa: encuentra puentes (aristas que al remover desconectan el grafo) usando tiempos de entrada y lowlink
vector<vector<int>> adj;
vector<int> tin, low;
vector<array<int,2>> ans;
int t = 1;

void dfs(int node, int parent = -1){
	tin[node] = low[node] = t++;
	for(auto adjN : adj[node]){
		if(adjN == parent) continue;
		if(!tin[adjN]){
			dfs(adjN, node);
			low[node] = min(low[node], low[adjN]);
			if(tin[node] < low[adjN]){
				int mn = min(node, adjN), mx = max(node, adjN);
				ans.push_back({mn, mx});
			}
		}else{
			low[node] = min(low[node], tin[adjN]);
		}
	}
}

void solve(){
	int n, m; cin >> n >> m;
	adj.assign(n + 1, {});
	tin.assign(n + 1, 0);
	low.assign(n + 1, 0);
	ans.clear();
	t = 1;

	for(int i = 0; i < m; i++){
		int u, v; cin >> u >> v;
		adj[u].push_back(v);
		adj[v].push_back(u);
	}

	for(int i = 1; i <= n; i++){
		if(!tin[i]) dfs(i, -1);
	}
}
\end{lstlisting}
