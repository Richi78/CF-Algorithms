\subsubsection{Prim (priority queue)}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O((N + M) log N)
// usa: MST alternativo, mejor en grafos densos
#include <bits/stdc++.h>
using namespace std;

const long long INF = 1e18;

long long prim(int n, const vector<vector<pair<int,int>>>& adj){
	vector<long long> dist(n, INF);
	vector<bool> vis(n, false);
	priority_queue<pair<long long,int>, vector<pair<long long,int>>, greater<pair<long long,int>>> pq;
	dist[0] = 0;
	pq.push({0, 0});
	long long total = 0;
	while(!pq.empty()){
		auto [dis, node] = pq.top(); pq.pop();
		if(vis[node]) continue;
		vis[node] = true;
		total += dis;
		for(auto [adjN, adjW] : adj[node]){
			if(!vis[adjN] && adjW < dist[adjN]){
				dist[adjN] = adjW;
				pq.push({adjW, adjN});
			}
		}
	}
	return total;
}

int main(){
	vector<vector<pair<int,int>>> adj(4);
	adj[0] = {{1,4},{3,5}}; adj[1] = {{0,4},{2,1},{3,2}};
	adj[2] = {{1,1},{3,2}}; adj[3] = {{0,5},{1,2},{2,2}};
	cout << "MST weight = " << prim(4, adj) << "\n";  // 7
	return 0;
}
\end{lstlisting}
