\subsubsection{Segment Tree (con BASE)}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(log N) por operacion, O(N) espacio
// usa: sin struct, BASE potencia de 2, update punto y query rango
#include <bits/stdc++.h>
using namespace std;

int BASE = 1;
vector<long long> st;

void build(int n, const vector<long long>& a = {}){
  BASE = 1;
  while(BASE < n) BASE <<= 1;
  st.assign(2 * BASE, 0); // o -INF para max
  for(int i = 0; i < (int)a.size(); i++) st[BASE + i] = a[i];
  for(int i = BASE - 1; i > 0; i--) st[i] = st[2 * i] + st[2 * i + 1];
}

// Update puntual: asigna val en posicion u
void update(int u, long long val){
  st[BASE + u] = val;
  for(int i = (BASE + u) / 2; i > 0; i >>= 1){
    st[i] = st[2 * i] + st[2 * i + 1];
  }
}

// 1. Consulta iterativa en rango [ql, qr]:
long long query_it(int ql, int qr){
  long long res = 0;
  for(int l = BASE + ql, r = BASE + qr + 1; l < r; l >>= 1, r >>= 1){
    if(l & 1) res += st[l++];
    if(r & 1) res += st[--r];
  }
  return res;
}

// 2. Consulta recursiva con l, r, ql, qr:
long long _get(int node, int l, int r, int ql, int qr){
  if(r < ql || qr < l) return 0;
  if(ql <= l && r <= qr) return st[node];
  int mid = l + (r - l) / 2;
  return _get(2 * node, l, mid, ql, qr) + 
         _get(2 * node + 1, mid + 1, r, ql, qr);
}

long long get(int ql, int qr){
  return _get(1, 0, BASE - 1, ql, qr);
}

int main(){
  int n = 5;
  vector<long long> a = {1, 2, 3, 4, 5};
  build(n, a);
  cout << get(1, 3) << "\n";      // 9 (2 + 3 + 4)
  update(2, 10);
  cout << query_it(1, 3) << "\n"; // 16 (2 + 10 + 4)
  return 0;
}
\end{lstlisting}
