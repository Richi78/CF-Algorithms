\subsubsection{Binary Lifting / LCA}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N log N) build, O(log N) por query
// usa: lowest common ancestor y k-esimo ancestro en arbol
#include <bits/stdc++.h>
using namespace std;

const int LOG = 20; // 2^20 > 1e6
vector<int> depth;
vector<vector<int>> up;

void dfs(int node, int par, int d, vector<int> adj[]){
  depth[node] = d;
  up[0][node] = par;
  for(auto adjN : adj[node]) if(adjN != par){
    dfs(adjN, node, d + 1, adj);
  }
}

// Camina 'step' pasos hacia arriba desde 'node' (k-th ancestor)
int walk(int node, int step){
  for(int i = 0; i <= LOG; i++) if((step >> i) & 1){
    node = up[i][node];
  }
  return node;
}

int get_lca(int u, int v){
  if(depth[u] > depth[v]) swap(u, v);
  v = walk(v, depth[v] - depth[u]);
  if(u == v) return u;
  for(int i = LOG; i >= 0; i--){
    if(up[i][u] != up[i][v]){
      u = up[i][u];
      v = up[i][v];
    }
  }
  return up[0][u];
}

int dist(int u, int v){
  return depth[u] + depth[v] - 2 * depth[get_lca(u, v)];
}

void init_lca(int root, int n, vector<int> adj[]){
  depth.assign(n, 0);
  up.assign(LOG + 1, vector<int>(n, 0));
  dfs(root, root, 0, adj);
  for(int i = 1; i <= LOG; i++){
    for(int j = 0; j < n; j++){
      up[i][j] = up[i - 1][up[i - 1][j]];
    }
  }
}
\end{lstlisting}
