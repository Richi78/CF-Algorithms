\subsubsection{Segment Tree 2D (con BASE)}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(log N * log M) por operacion
// usa: sin struct, BASE_X y BASE_Y potencias de 2, queries en rectangulo
#include <bits/stdc++.h>
using namespace std;

int BASE_X = 1, BASE_Y = 1;
vector<vector<int>> st;

void build(int n, int m, const vector<vector<int>>& a = {}){
  BASE_X = 1; while(BASE_X < n) BASE_X <<= 1;
  BASE_Y = 1; while(BASE_Y < m) BASE_Y <<= 1;
  st.assign(2 * BASE_X, vector<int>(2 * BASE_Y, 0));

  if(!a.empty()){
    for(int i = 0; i < n; i++)
      for(int j = 0; j < m; j++)
        st[BASE_X + i][BASE_Y + j] = a[i][j];

    // Construir arboles en dimension Y
    for(int i = BASE_X; i < 2 * BASE_X; i++)
      for(int j = BASE_Y - 1; j > 0; j--)
        st[i][j] = st[i][2 * j] + st[i][2 * j + 1];

    // Construir arboles en dimension X
    for(int i = BASE_X - 1; i > 0; i--)
      for(int j = 1; j < 2 * BASE_Y; j++)
        st[i][j] = st[2 * i][j] + st[2 * i + 1][j];
  }
}

// Update puntual en celda (x, y) con val
void update(int x, int y, int val){
  int px = BASE_X + x, py = BASE_Y + y;
  st[px][py] = val;
  for(int j = py / 2; j > 0; j >>= 1){
    st[px][j] = st[px][2 * j] + st[px][2 * j + 1];
  }
  for(int i = px / 2; i > 0; i >>= 1){
    for(int j = py; j > 0; j >>= 1){
      st[i][j] = st[2 * i][j] + st[2 * i + 1][j];
    }
  }
}

// Consultas recursivas con ly, ry, qly, qry y lx, rx, qlx, qrx
int _query_y(int nodex, int nodey, int ly, int ry, int qly, int qry){
  if(ry < qly || qry < ly) return 0;
  if(qly <= ly && ry <= qry) return st[nodex][nodey];
  int mid = ly + (ry - ly) / 2;
  return _query_y(nodex, 2 * nodey, ly, mid, qly, qry) +
         _query_y(nodex, 2 * nodey + 1, mid + 1, ry, qly, qry);
}

int _query_x(int nodex, int lx, int rx, int qlx, int qrx, int qly, int qry){
  if(rx < qlx || qrx < lx) return 0;
  if(qlx <= lx && rx <= qrx){
    return _query_y(nodex, 1, 0, BASE_Y - 1, qly, qry);
  }
  int mid = lx + (rx - lx) / 2;
  return _query_x(2 * nodex, lx, mid, qlx, qrx, qly, qry) +
         _query_x(2 * nodex + 1, mid + 1, rx, qlx, qrx, qly, qry);
}

int query(int qlx, int qrx, int qly, int qry){
  return _query_x(1, 0, BASE_X - 1, qlx, qrx, qly, qry);
}

int main(){
  build(4, 4);
  update(1, 1, 5);
  update(2, 2, 7);
  cout << query(0, 3, 0, 3) << "\n"; // 12
  cout << query(1, 1, 1, 1) << "\n"; // 5
  return 0;
}
\end{lstlisting}
