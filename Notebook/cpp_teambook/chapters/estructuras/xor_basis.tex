\subsubsection{Basis (XOR basis)}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(D^2) por insercion (D = numero de bits, aca 64), O(D^2) total para n inserciones
// usa: representar conjunto de numeros como base de Gauss sobre GF(2); maximo xor subconjunto, pertenencia, k-esimo xor
struct Basis{
	int d=64;
	vector<int> basis;
	int sz=0; // current size of the basis
	Basis(){
		basis.resize(d);
	}
	void insertVector(int mask){
		for(int i=d-1 ; i>=0 ; i--){
			if( (mask & (1LL<<i)) == 0 ) continue;
			if( !basis[i] ){
				basis[i]=mask;
				++sz;
				return;
			}
			mask ^= basis[i];
		}
	}
};

int main(){
	// ejemplo: dado un conjunto de numeros, encontrar el maximo xor de un subconjunto
	vector<int> nums = {1, 2, 3, 5};
	Basis b;
	for(int x : nums) b.insertVector(x);
	int ans = 0;
	for(int i=b.d-1 ; i>=0 ; i--) ans = max(ans, ans ^ b.basis[i]);
	// ans = maximo xor de un subconjunto de nums
	return 0;
}
\end{lstlisting}
