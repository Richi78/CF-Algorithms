\subsubsection{Random / mt19937}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(1) per call
// usa: shuffling deterministico, generacion de casos, hash randomization
#include <bits/stdc++.h>
using namespace std;

// Seed with fixed value for reproducible runs; change chrono for varied.
mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());

// integer in [lo, hi]
int randInt(int lo, int hi){
	return uniform_int_distribution<int>(lo, hi)(rng);
}

// real in [lo, hi)
double randReal(double lo, double hi){
	return uniform_real_distribution<double>(lo, hi)(rng);
}

// shuffle a vector in O(n)
template<class T>
void shuffleVec(vector<T>& v){
	shuffle(v.begin(), v.end(), rng);
}

// pick a random element from a vector (non-empty)
template<class T>
T& randPick(vector<T>& v){
	return v[uniform_int_distribution<int>(0, (int)v.size() - 1)(rng)];
}

int main(){
	// shuffle a vector deterministically
	vector<int> a = {1,2,3,4,5};
	shuffleVec(a);
	for(int x : a) cout << x << " ";
	cout << "\n";

	// random int in [lo, hi]
	int r = randInt(1, 100);
	cout << "randInt = " << r << "\n";

	// shuffle with a fixed seed for reproducibility (replace rng temporarily)
	mt19937_64 fixedRng(42);
	shuffle(a.begin(), a.end(), fixedRng);
	for(int x : a) cout << x << " ";
	cout << "\n";
	return 0;
}
\end{lstlisting}
