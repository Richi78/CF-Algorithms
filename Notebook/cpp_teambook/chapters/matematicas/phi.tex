\subsubsection{Euler totient precompute}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N log log N)
// usa: queries tipo "phi(n) para varios n hasta N"
#include <bits/stdc++.h>
using namespace std;

vector<int> phi;

void phiSieve(int n){
	phi.assign(n + 1, 0);
	for(int i=0;i<=n;i++) phi[i] = i;
	for(int i=2;i<=n;i++){
		if(phi[i] == i){ // prime
			for(int j=i;j<=n;j+=i){
				phi[j] -= phi[j] / i;
			}
		}
	}
}

// single phi via factorization (slower but no precompute)
long long phiSingle(long long n){
	long long result = n;
	for(long long p=2 ; p*p<=n ; p++){
		if(n % p == 0){
			while(n % p == 0) n /= p;
			result -= result / p;
		}
	}
	if(n > 1) result -= result / n;
	return result;
}

int main(){
	phiSieve(100);
	// phi[10] = 4 (numbers coprime with 10: 1,3,7,9)
	for(int i=1 ; i<=10 ; i++) 
		cout << "phi(" << i << ") = " << phi[i] << "\n";
	// single phi
	cout << "phi(36) = " << phiSingle(36) << "\n";
	return 0;
}
\end{lstlisting}
