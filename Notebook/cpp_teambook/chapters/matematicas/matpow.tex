\subsubsection{Matrix exponentiation}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N^3 log k)
// usa: recurrencias lineales (Fibonacci, contar caminos de longitud k, etc.)
#include <bits/stdc++.h>
using namespace std;

using Matrix = vector<vector<long long>>;

Matrix matMul(const Matrix& a, const Matrix& b, long long MOD){
	int n = a.size();
	Matrix c(n, vector<long long>(n, 0));
	for(int i=0;i<n;i++)
		for(int k=0;k<n;k++) if(a[i][k])
			for(int j=0;j<n;j++)
				c[i][j] = (c[i][j] + a[i][k] * b[k][j]) % MOD;
	return c;
}

Matrix matPow(Matrix base, long long exp, long long MOD){
	int n = base.size();
	Matrix res(n, vector<long long>(n, 0));
	for(int i=0;i<n;i++) res[i][i] = 1;
	while(exp){
		if(exp & 1) res = matMul(res, base, MOD);
		base = matMul(base, base, MOD);
		exp >>= 1;
	}
	return res;
}

int main(){
	// Fibonacci: F(n) = F(n-1) + F(n-2)
	Matrix T = {{1,1},{1,0}};
	Matrix Fn = matPow(T, 10, MOD);
	cout << "F(10) = " << Fn[0][1] << "\n";  // 55
	return 0;
}
\end{lstlisting}
