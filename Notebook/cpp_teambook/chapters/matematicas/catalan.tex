\subsubsection{Catalan (formula)}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(n^2)
vector<int> dp;
const int MOD=998244353;

void catalan(int n){ // from 1 to n
	dp.resize(n+1);
	dp[0]=dp[1]=1;
	for(int i=2 ; i<=n ; i++){
		for(int j=0 ; j<i ; j++){
			dp[i]=(dp[i]+(dp[j]*dp[i-j-1])%MOD)%MOD;
		}
	}
}

// TC: O(n)
// consider use of long long
int catalan(int n){
	int ans=1;
	for(int i=0 ; i<n ; i++){
		ans=ans*2*(2*i+1)/(i+2);
	}
	return ans;
}

int main(){
	cout << "catalan(10) = " << catalan(10) << "\n";
	return 0;
}
\end{lstlisting}
