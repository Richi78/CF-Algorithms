\subsubsection{Mobius precompute}
\begin{lstlisting}[style=mystyle, language=C++]
// TC: O(N log log N)
// usa: inclusion-exclusion con Mobius, queries tipo sum_{d|n} mu(d)
#include <bits/stdc++.h>
using namespace std;

vector<int> mu;

void mobiusSieve(int n){
	mu.assign(n + 1, 1);
	vector<bool> is_composite(n + 1, false);
	vector<int> primes;
	mu[0] = 0;
	mu[1] = 1;
	for(int i=2;i<=n;i++){
		if(!is_composite[i]){
			primes.push_back(i);
			mu[i] = -1;
		}
		for(int p : primes){
			long long v = 1LL * i * p;
			if(v > n) break;
			is_composite[v] = true;
			if(i % p == 0){
				mu[v] = 0;
				break;
			}else{
				mu[v] = -mu[i];
			}
		}
	}
}

int main(){
	mobiusSieve(20);
	for(int i=1 ; i<=10 ; i++) 
		cout << "mu(" << i << ") = " << mu[i] << "\n";
	return 0;
}
\end{lstlisting}
