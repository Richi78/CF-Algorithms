\subsubsection{Formulas utiles}
Numero de divisores
\[
d(n) = (e_1 + 1) \cdot (e_2 + 1) \cdot \dots \cdot (e_k + 1)
\]

Suma de divisores
\[
\sigma(n) = \frac{p_1^{e_1 + 1} - 1}{p_1 - 1} \cdot
\frac{p_2^{e_2 + 1} - 1}{p_2 - 1} \cdot
\dots \cdot
\frac{p_k^{e_k + 1} - 1}{p_k - 1}
\]

Producto de divisores
\[
P(n) = n^{\frac{d(n)}{2}}
\]

Euler
\[
a^b \equiv a^{\,b \bmod \varphi(n)} \pmod{n}
\]

Inclusion -- Exclusion
\[
\left| \bigcup_{i=1}^{n} A_i \right|
= \sum_{\emptyset \neq I \subseteq \{1,2,\dots,n\}} (-1)^{|I|+1}
\left| \bigcap_{i \in I} A_i \right|
\]
